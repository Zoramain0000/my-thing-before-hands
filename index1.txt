async function fetchYouTube() {

            );

// 🎨 ANIMATE COUNTER
// ═══════════════════════════════════════════════════════════════════
function anim(el, target, dur = 800) {
    const s = parseInt(el.textContent.replace(/,/g, '')) || 0;
    if (s === target) { el.textContent = target.toLocaleString(); return; }
    const t0 = performance.now();
    function u(t) {
        const p = Math.min((t - t0) / dur, 1);
        const e = 1 - Math.pow(1 - p, 3);
        el.textContent = Math.round(s + (target - s) * e).toLocaleString();
        if (p < 1) requestAnimationFrame(u);
    }
    requestAnimationFrame(u);
}

                    setTimeout(() => el.classList.remove('count-flash'), 600);

    const fbb = document.querySelector('[data-bar="fb"]');
    if (fbb) fbb.style.width = Math.min((fb / P.fb.goal) * 100, 100) + '%';
    const ttb = document.querySelector('[data-bar="tt"]');

    // Delta indicators

        const el = document.getElementById(id);
        if (!el) return;
        if (prev !== null && prev !== curr) {
            const diff = curr - prev;
            if (diff > 0) { el.textContent = `+${diff}`; el.className = 'text-[9px] text-green-400 font-semibold'; }
            else if (diff < 0) { el.textContent = `${diff}`; el.className = 'text-[9px] text-red-400 font-semibold'; }
            else { el.textContent = ''; el.className = 'text-[9px] text-neutral-600'; }
        }

    setDelta('fbDelta', P.fb.prev, P.fb.count);
    setDelta('ttDelta', P.tt.prev, P.tt.count);
